Undo/redo

How it works

A process is a function, so middleware is function composition — no store enhancers, no plugin API. The wrapper hands the inner process a private mailbox (channel) and owns the history itself. The same shape gives logging, persistence, and replay.

type Hist = { type: 'undo' } | { type: 'redo' }

function withHistory<T, In, A>(proc: Proc<T, In, A>, depth = 100): Proc<T, In | Hist, A> {
  return async function* (self, args) {
    const inner = channel<In>(self.signal)   // a private Self for the wrapped proc
    const it = proc(inner, args)
    let current = (await it.next()).value as T
    const past: T[] = []
    const future: T[] = []
    yield current
    for await (const msg of self) {
      if ((msg as Hist).type === 'undo' && past.length > 0) {
        future.push(current)
        current = past.pop() as T
      } else if ((msg as Hist).type === 'redo' && future.length > 0) {
        past.push(current)
        current = future.pop() as T
      } else {
        past.push(current)
        inner.cast(msg as In)
        current = (await it.next()).value as T
      }
      yield current
    }
  }
}
